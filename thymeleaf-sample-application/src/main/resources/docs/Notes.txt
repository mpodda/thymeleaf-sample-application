import org.thymeleaf.context.WebContext;
import org.thymeleaf.web.servlet.JakartaServletWebApplication;
import org.thymeleaf.web.servlet.JakartaServletWebExchange;


public String render(HttpServletRequest request,
                     HttpServletResponse response) {

    var application = JakartaServletWebApplication.buildApplication(request.getServletContext());
    var exchange = new JakartaServletWebExchange(request, response, application);

    WebContext context = new WebContext(exchange);
    context.setVariable("name", "John");

    return templateEngine.process("template", context);
}
